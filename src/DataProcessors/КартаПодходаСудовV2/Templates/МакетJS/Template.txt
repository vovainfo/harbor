    function hasDimensions(vessel) {
        return vessel.toBow != null && vessel.toStern != null && vessel.toPort != null

    // Пятиугольник корпуса в координатах судна (x — к правому борту, y — к носу), повёрнутый по курсу.
    function hullPolygon(vessel, rot) {
        var a = vessel.toBow;
        var b = vessel.toStern;
        var c = vessel.toPort;
        var d = vessel.toStarboard;
        var shoulder = a - 0.15 * (a + b);
        var local = [[-c, -b], [d, -b], [d, shoulder], [(d - c) / 2, a], [-c, shoulder]];
        var theta = rot * Math.PI / 180;
        var cos = Math.cos(theta);
        var sin = Math.sin(theta);
        var metersPerDegLon = 111320 * Math.cos(vessel.lat * Math.PI / 180);
        var ring = local.map(function (point) {
            var east = point[0] * cos + point[1] * sin;
            var north = -point[0] * sin + point[1] * cos;
            return [vessel.lon + east / metersPerDegLon, vessel.lat + north / 111320];
        });
        ring.push(ring[0]);
        return ring;
    }

    function matchesWatch(item, vessel) {
        if (item.imo && vessel.imo) {
            return item.imo === vessel.imo;

        return item.mmsi != null && item.mmsi === vessel.mmsi;
    }

    function isWatched(vessel) {
        return (watchlist.vessels || []).some(function (item) { return matchesWatch(item, vessel); });
    }

    function rebuildVesselData() {
        var points = [];
        var hulls = [];
        vesselsByMmsi = {};
        var vessels = snapshot && snapshot.vessels ? snapshot.vessels : [];
        var receivedAt = snapshot ? Date.parse(snapshot.receivedAt) : NaN;

        vessels.forEach(function (vessel) {
            var rot = rotationOf(vessel);
            var ageMin = Math.max(0, (receivedAt - Date.parse(vessel.seen)) / 60000);
            if (!isFinite(ageMin)) {
                ageMin = 0;
            }
            var group = groupOf(vessel.shipType);
            var hasHull = rot != null && hasDimensions(vessel);
            var watched = isWatched(vessel);
            vesselsByMmsi[vessel.mmsi] = vessel;
            points.push({
                type: 'Feature',
                geometry: { type: 'Point', coordinates: [vessel.lon, vessel.lat] },
                properties: {
                    mmsi: vessel.mmsi,
                    group: group,
                    moving: vessel.sog != null && vessel.sog >= 1,
                    rot: rot,
                    label: vessel.name ? vessel.name : String(vessel.mmsi),
                    ageMin: ageMin,
                    ageClass: ageClassOf(ageMin),
                    hasHull: hasHull,
                    watched: watched
                }
            });
            if (hasHull) {
                hulls.push({
                    type: 'Feature',
                    geometry: { type: 'Polygon', coordinates: [hullPolygon(vessel, rot)] },
                    // watched и у корпуса: фильтр и цвет контура читают свойства корпуса.
                    properties: {
                        mmsi: vessel.mmsi, group: group, ageMin: ageMin, ageClass: ageClassOf(ageMin), watched: watched
                    }
                });
            }
        });
        vesselData = {
            points: { type: 'FeatureCollection', features: points },
            hulls: { type: 'FeatureCollection', features: hulls }
        };
    }

    function drawVesselShape(ctx, size, shape, color) {
        var s = size;
        ctx.beginPath();
        if (shape === 'arrow') {
            // Нос смотрит вверх: при icon-rotate = 0 судно идёт на север.
            ctx.moveTo(s * 0.5, s * 0.06);
            ctx.lineTo(s * 0.82, s * 0.9);
            ctx.lineTo(s * 0.5, s * 0.72);
            ctx.lineTo(s * 0.18, s * 0.9);
        } else {
            ctx.moveTo(s * 0.5, s * 0.2);
            ctx.lineTo(s * 0.78, s * 0.5);
            ctx.lineTo(s * 0.5, s * 0.8);
            ctx.lineTo(s * 0.22, s * 0.5);
        }
        ctx.closePath();
        ctx.fillStyle = color;
        ctx.fill();
        ctx.lineWidth = Math.max(1, s * 0.07);
        ctx.strokeStyle = '#ffffff';
        ctx.stroke();
    }

    function vesselIconCanvas(shape, color, cssSize, ratio) {
        var canvas = document.createElement('canvas');
        var size = Math.round(cssSize * ratio);
        canvas.width = size;

        return canvas;

            ['arrow', 'diamond'].forEach(function (shape) {
                var name = group.id + '-' + shape;
                if (map.hasImage(name)) {

                var image = canvas.getContext('2d').getImageData(0, 0, canvas.width, canvas.height);
                map.addImage(name, image, { pixelRatio: ratio });
            });
        });
    }

    // Видимость, фильтр возраста и выделение — из переменных viewer, поэтому переживают смену стиля.
    function applyLayerState() {
        SEAMARK_LAYER_IDS.forEach(function (layerId) {
            setVisibility(layerId, ui.seamark);
        });
        setVisibility('vessel-hulls-fill', ui.hulls);
        setVisibility('vessel-hulls-line', ui.hulls);
        setVisibility('vessel-hulls-selected', ui.hulls);
        setVisibility('vessel-labels', ui.labels);
        setVisibility('vessel-labels-watched', ui.labels);
        AGE_OPACITY_PAINT.forEach(function (paint) {
            setPaintIfPresent(paint[0], paint[1], ageOpacityExpression(paint[2]));
        });
        document.getElementById('legend-age').style.display = ui.ageFade ? '' : 'none';

        var ageFilter = ui.maxAgeMin == null ? null : ['<=', ['get', 'ageMin'], ui.maxAgeMin];
        var watchedFilter = ['==', ['get', 'watched'], true];
        var onlyWatched = ui.watchedOnly ? watchedFilter : null;
        // Общие условия слоя: возраст, «Только отслеживаемые» и своё условие слоя.
        function compose(conditions) {
            var parts = conditions.filter(function (part) { return !!part; });
            if (!parts.length) {
                return null;
            }
            return parts.length === 1 ? parts[0] : ['all'].concat(parts);
        }
        setFilterIfPresent('vessel-icons', compose([ageFilter, onlyWatched]));
        // При выключенных корпусах значок нужен всем судам и на крупном масштабе.
        setFilterIfPresent('vessel-icons-hull-zoom',
            compose([ageFilter, onlyWatched, ui.hulls ? ['!=', ['get', 'hasHull'], true] : null]));
        setFilterIfPresent('vessel-labels', compose([ageFilter, onlyWatched, ['!=', ['get', 'watched'], true]]));
        setFilterIfPresent('vessel-labels-watched', compose([ageFilter, watchedFilter]));
        setFilterIfPresent('vessel-watched-ring', compose([ageFilter, watchedFilter]));
        setFilterIfPresent('vessel-hulls-fill', compose([ageFilter, onlyWatched]));
        setFilterIfPresent('vessel-hulls-line', compose([ageFilter, onlyWatched]));

        var selection = ['==', ['get', 'mmsi'], selectedMmsi];
        setFilterIfPresent('vessel-selected', compose([selection, onlyWatched]));
        setFilterIfPresent('vessel-hulls-selected', compose([selection, onlyWatched]));
    }

    function escapeHtml(text) {
        return String(text).replace(/[&<>"']/g, function (ch) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[ch];
        });
    }

    // Атрибуция снимка приходит из внешней сети: в контрол она попадает только экранированной.

        var text = snapshot && snapshot.attribution ? snapshot.attribution : 'Open Waters AIS';
        if (text === attributionText) {
            return;

        attributionText = text;
        if (attributionControl) {
            map.removeControl(attributionControl);

        attributionControl = new maplibregl.AttributionControl({
            compact: true,
            customAttribution: escapeHtml(text)
        });
        map.addControl(attributionControl, 'bottom-right');
    }

    function updateStaleBanner() {

    function formatAge(ageMin) {
        var minutes = Math.max(0, Math.floor(ageMin));
        if (minutes < 60) {
            return minutes + ' мин';
        }
        var hours = Math.floor(minutes / 60);
        if (hours < 24) {
            return hours + ' ч ' + (minutes % 60) + ' мин';
        }
        return Math.floor(hours / 24) + ' сут ' + (hours % 24) + ' ч';
    }

    function formatNumber(value, digits, unit) {
        return value == null ? '—' : Number(value).toFixed(digits) + (unit || '');
    }

    function openPopup(mmsi) {
        var vessel = vesselsByMmsi[mmsi];

    // Позиция «старше 1 ч» — от времени снимка, той же шкалой, что легенда: устаревший снимок целиком
    // уже отмечает жёлтая плашка.
    function watchStatus(item) {
        var vessels = snapshot && snapshot.vessels ? snapshot.vessels : [];
        var receivedAt = snapshot ? Date.parse(snapshot.receivedAt) : NaN;
        for (var i = 0; i < vessels.length; i++) {
            if (matchesWatch(item, vessels[i])) {
                var ageMin = (receivedAt - Date.parse(vessels[i].seen)) / 60000;
                return {
                    vessel: vessels[i],
                    code: ageMin > WATCH_OLD_POSITION_MIN ? 'old' : 'on-map',
                    text: ageMin > WATCH_OLD_POSITION_MIN ? 'позиция старше 1 ч' : 'на карте'
                };
            }
        }
        return { vessel: null, code: 'missing', text: 'нет в снимке' };
    }

    function seamarkColour(tags, type) {
        var value = tags['seamark:' + type + ':colour'] || tags['seamark:light:colour'] || tags['seamark:light:1:colour'];
        return value ? SEAMARK_COLOURS[String(value).split(';')[0]] || null : null;
    }

    function hasLight(tags, type) {
        if (type.indexOf('light') === 0) {
            return true;
        }
        return Object.keys(tags).some(function (key) { return key.indexOf('seamark:light') === 0; });
    }

    // Цвета буя или знака сверху вниз (до трёх полос); имена — ключи SEAMARK_COLOURS.
    function seamarkBands(tags, type) {
        var value = tags['seamark:' + type + ':colour'];
        var bands = value ? String(value).split(';').filter(function (c) { return !!SEAMARK_COLOURS[c]; }) : [];
        return bands.length ? bands.slice(0, 3).join('.') : 'grey';
    }

    // Белый и жёлтый огонь на карте — жёлтая «капля», красный и зелёный — своим цветом.
    function lightColour(tags) {
        var value = tags['seamark:light:colour'] || tags['seamark:light:1:colour'];
        var colour = value ? String(value).split(';')[0] : '';
        return colour === 'red' || colour === 'green' ? colour : 'yellow';
    }

    // Имя значка кодирует всё, что нужно для рисования: sm-<вид>-<форма>-<цвета через точку>.
    // Картинку рисует addSeamarkIcon по событию styleimagemissing.
    function seamarkIcon(tags, type) {
        if (type === 'anchorage' || type === 'anchor_berth') {
            return 'sm-anchor';
        }
        if (type === 'harbour') {
            return 'sm-harbour';
        }
        if (type.indexOf('light') === 0) {
            return 'sm-light-' + lightColour(tags);
        }
        if (type.indexOf('buoy_') === 0) {
            var shape = String(tags['seamark:' + type + ':shape'] || '').replace(/[^a-z]/g, '').replace('superbuoy', 'super');
            return 'sm-buoy-' + (SEAMARK_BUOY_SHAPES.indexOf(shape) === -1 ? 'pillar' : shape) + '-' + seamarkBands(tags, type);
        }
        if (type.indexOf('beacon_') === 0) {
            return 'sm-beacon-' + seamarkBands(tags, type);
        }
        if (type === 'wreck') {
            return 'sm-wreck';
        }
        if (type === 'rock' || type === 'obstruction') {
            return 'sm-danger';
        }
        if (type === 'pilot_boarding') {
            return 'sm-pilot';
        }
        if (type === 'radio_station') {
            return 'sm-radio';
        }
        return null;
    }

    function seamarkProperties(tags) {
        var type = tags['seamark:type'];
        var properties = { type: type, light: hasLight(tags, type) };
        var name = tags['name:ru'] || tags['seamark:name'] || tags.name;
        if (name) {
            properties.name = name;
        }
        var colour = seamarkColour(tags, type);
        if (colour) {
            properties.colour = colour;
        }
        var icon = seamarkIcon(tags, type);
        if (icon) {
            properties.icon = icon;
            // Буй и знак стоят основанием на точке, остальные значки — центром.
            properties.iconAnchor = icon.indexOf('sm-buoy') === 0 || icon.indexOf('sm-beacon') === 0 ? 'bottom' : 'center';
        }
        return properties;
    }

    // Центр масс кольца; для мультиполигона — самого большого кольца.
    function ringCentroid(ring) {
        var area = 0;
        var x = 0;
        var y = 0;
        for (var i = 0; i < ring.length - 1; i++) {
            var cross = ring[i][0] * ring[i + 1][1] - ring[i + 1][0] * ring[i][1];
            area += cross;
            x += (ring[i][0] + ring[i + 1][0]) * cross;
            y += (ring[i][1] + ring[i + 1][1]) * cross;
        }
        return area ? { point: [x / (3 * area), y / (3 * area)], area: Math.abs(area) } : { point: ring[0], area: 0 };
    }

    function areaCentre(geometry) {
        var rings = geometry.type === 'Polygon' ? [geometry.coordinates[0]]
            : geometry.coordinates.map(function (polygon) { return polygon[0]; });
        var best = null;
        rings.forEach(function (ring) {
            var centroid = ringCentroid(ring);
            if (!best || centroid.area > best.area) {
                best = centroid;
            }
        });
        return best.point;
    }

    function toLonLat(points) {
        return points.map(function (point) { return [point.lon, point.lat]; });
    }

    function samePoint(a, b) {
        return a[0] === b[0] && a[1] === b[1];
    }

    function isClosed(line) {
        return line.length >= 4 && samePoint(line[0], line[line.length - 1]);
    }

    // Внешние пути мультиполигона Overpass отдаёт кусками: склеиваем их по общим концам в кольца.
    function joinRings(segments) {
        var rest = segments.slice();
        var rings = [];
        while (rest.length) {
            var ring = rest.shift().slice();
            var extended = true;
            while (!isClosed(ring) && extended) {
                extended = false;
                for (var i = 0; i < rest.length; i++) {
                    var segment = rest[i];
                    var last = ring[ring.length - 1];
                    if (samePoint(segment[0], last)) {
                        ring = ring.concat(segment.slice(1));
                    } else if (samePoint(segment[segment.length - 1], last)) {
                        ring = ring.concat(segment.slice(0, -1).reverse());
                    } else {
                        continue;
                    }
                    rest.splice(i, 1);
                    extended = true;
                    break;
                }
            }
            if (isClosed(ring)) {
                rings.push(ring);
            }
        }
        return rings;
    }

    function seamarkGeometry(element, properties) {
        if (element.type === 'node') {
            return { type: 'Point', coordinates: [element.lon, element.lat] };
        }
        if (element.type === 'way' && element.geometry && element.geometry.length >= 2) {
            var line = toLonLat(element.geometry);
            if (isClosed(line) && SEAMARK_AREA_TYPES.indexOf(properties.type) !== -1) {
                return { type: 'Polygon', coordinates: [line] };
            }
            return { type: 'LineString', coordinates: line };
        }
        if (element.type === 'relation' && element.members) {
            var outers = element.members.filter(function (member) {
                return member.type === 'way' && member.role !== 'inner' && member.geometry && member.geometry.length >= 2;
            });
            var rings = joinRings(outers.map(function (member) { return toLonLat(member.geometry); }));
            if (rings.length) {
                return { type: 'MultiPolygon', coordinates: rings.map(function (ring) { return [ring]; }) };
            }
        }
        return null;
    }

    // Ответ Overpass (out geom) → GeoJSON. Узлы — точки, замкнутые пути площадных типов — районы,
    // остальные пути — линии, мультиполигоны — только внешние кольца.
    function buildSeamarkData() {
        var overpass = harborCache.json('seamark/overpass');
        if (!overpass) {
            return null;
        }
        var features = [];
        (overpass.elements || []).forEach(function (element) {
            var tags = element.tags || {};
            if (!tags['seamark:type']) {
                return;
            }
            var properties = seamarkProperties(tags);
            var geometry = seamarkGeometry(element, properties);
            if (!geometry) {
                return;
            }
            features.push({ type: 'Feature', geometry: geometry, properties: properties });
            // Значок района — отдельной точкой в центре: заливка района значка не несёт.
            var area = geometry.type === 'Polygon' || geometry.type === 'MultiPolygon' || isClosed(geometry.coordinates);
            if (properties.icon && area && SEAMARK_CENTER_TYPES.indexOf(properties.type) !== -1) {
                features.push({
                    type: 'Feature',
                    geometry: { type: 'Point', coordinates: areaCentre(geometry.type === 'LineString'
                        ? { type: 'Polygon', coordinates: [geometry.coordinates] } : geometry) },
                    properties: { type: properties.type, icon: properties.icon, iconAnchor: properties.iconAnchor }
                });
            }
        });
        return { type: 'FeatureCollection', features: features };
    }

    // ---------- Значки морской обстановки ----------

    function strokeWithHalo(ctx, colour, width) {
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.9)';
        ctx.lineWidth = width + 2.5;
        ctx.stroke();
        ctx.strokeStyle = colour;
        ctx.lineWidth = width;
        ctx.stroke();
    }

    function drawAnchor(ctx, w, h, colour) {
        var cx = w / 2;
        var cy = h / 2;
        var r = w * 0.42;
        var from = 0.15 * Math.PI;
        var to = 0.85 * Math.PI;
        ctx.beginPath();
        ctx.moveTo(cx + r * 0.18, cy - r * 0.75);
        ctx.arc(cx, cy - r * 0.75, r * 0.18, 0, 2 * Math.PI);
        ctx.moveTo(cx, cy - r * 0.57);
        ctx.lineTo(cx, cy + r * 0.85);
        ctx.moveTo(cx - r * 0.45, cy - r * 0.32);
        ctx.lineTo(cx + r * 0.45, cy - r * 0.32);
        ctx.moveTo(cx + r * 0.65 * Math.cos(from), cy + r * 0.2 + r * 0.65 * Math.sin(from));
        ctx.arc(cx, cy + r * 0.2, r * 0.65, from, to);
        strokeWithHalo(ctx, colour, 1.6);
    }

    // «Капля» огня по морским картам: острый конец на точке огня, тело уходит вправо вверх.
    function drawLight(ctx, w, h, colourName) {
        var length = w * 0.4;
        var radius = w * 0.15;
        ctx.save();
        ctx.translate(w / 2, h / 2);
        ctx.rotate(Math.PI / 4);
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.quadraticCurveTo(-radius * 1.1, -length * 0.55, -radius, -length);
        ctx.arc(0, -length, radius, Math.PI, 2 * Math.PI);
        ctx.quadraticCurveTo(radius * 1.1, -length * 0.55, 0, 0);
        ctx.closePath();
        ctx.fillStyle = colourName === 'yellow' ? '#ffc400' : SEAMARK_COLOURS[colourName];
        ctx.fill();
        ctx.lineWidth = 1;
        ctx.strokeStyle = SEAMARK_DARK;
        ctx.stroke();
        ctx.restore();
    }

    function buoyPath(ctx, shape, w, base, h) {
        var cx = w / 2;
        ctx.beginPath();
        if (shape === 'can') {
            ctx.rect(cx - w * 0.3, base - h * 0.5, w * 0.6, h * 0.5);
        } else if (shape === 'conical') {
            ctx.moveTo(cx - w * 0.34, base);
            ctx.lineTo(cx, base - h * 0.62);
            ctx.lineTo(cx + w * 0.34, base);
            ctx.closePath();
        } else if (shape === 'spherical') {
            ctx.arc(cx, base - w * 0.32, w * 0.32, 0, 2 * Math.PI);
        } else if (shape === 'spar') {
            ctx.rect(cx - w * 0.09, base - h * 0.8, w * 0.18, h * 0.8);
        } else if (shape === 'barrel') {
            ctx.ellipse(cx, base - h * 0.2, w * 0.4, h * 0.2, 0, 0, 2 * Math.PI);
        } else if (shape === 'super') {
            ctx.rect(cx - w * 0.42, base - h * 0.4, w * 0.84, h * 0.4);
        } else {
            ctx.moveTo(cx - w * 0.26, base);
            ctx.lineTo(cx - w * 0.1, base - h * 0.75);
            ctx.lineTo(cx + w * 0.1, base - h * 0.75);
            ctx.lineTo(cx + w * 0.26, base);
            ctx.closePath();
        }
    }

    // Полосы цветов сверху вниз внутри контура, затем контур и линия воды.
    function fillBands(ctx, w, h, bands) {
        var colours = bands.split('.');
        ctx.save();
        ctx.clip();
        for (var i = 0; i < colours.length; i++) {
            ctx.fillStyle = SEAMARK_COLOURS[colours[i]] || SEAMARK_COLOURS.grey;
            ctx.fillRect(0, h * i / colours.length, w, h / colours.length + 1);
        }
        ctx.restore();
        ctx.lineWidth = 1;
        ctx.strokeStyle = SEAMARK_DARK;
        ctx.stroke();
    }

    function drawBuoy(ctx, w, h, shape, bands) {
        var base = h - 2.5;
        buoyPath(ctx, shape, w, base, h - 4);
        fillBands(ctx, w, h, bands);
        ctx.beginPath();
        ctx.moveTo(w * 0.08, base + 1.2);
        ctx.lineTo(w * 0.92, base + 1.2);
        ctx.stroke();
    }

    function drawBeacon(ctx, w, h, bands) {
        var base = h - 1.5;
        ctx.beginPath();
        ctx.rect(w / 2 - w * 0.12, h * 0.1, w * 0.24, base - h * 0.1);
        fillBands(ctx, w, h, bands);
        ctx.beginPath();
        ctx.moveTo(w * 0.2, base);
        ctx.lineTo(w * 0.8, base);
        ctx.stroke();
    }

    function drawWreck(ctx, w, h) {
        var cy = h / 2;
        ctx.beginPath();
        ctx.moveTo(w * 0.12, cy);
        ctx.lineTo(w * 0.88, cy);
        ctx.moveTo(w * 0.32, cy - h * 0.18);
        ctx.lineTo(w * 0.32, cy + h * 0.18);
        ctx.moveTo(w * 0.5, cy - h * 0.28);
        ctx.lineTo(w * 0.5, cy + h * 0.28);
        ctx.moveTo(w * 0.68, cy - h * 0.18);
        ctx.lineTo(w * 0.68, cy + h * 0.18);
        strokeWithHalo(ctx, SEAMARK_DARK, 1.4);
    }

    function drawDanger(ctx, w, h) {
        ctx.beginPath();
        ctx.moveTo(w * 0.25, h / 2);
        ctx.lineTo(w * 0.75, h / 2);
        ctx.moveTo(w / 2, h * 0.25);
        ctx.lineTo(w / 2, h * 0.75);
        strokeWithHalo(ctx, SEAMARK_DARK, 1.6);
    }

    function drawLetterInCircle(ctx, w, h, letter) {
        ctx.beginPath();
        ctx.arc(w / 2, h / 2, w * 0.36, 0, 2 * Math.PI);
        ctx.fillStyle = 'rgba(255, 255, 255, 0.85)';
        ctx.fill();
        ctx.lineWidth = 1.4;
        ctx.strokeStyle = SEAMARK_MAGENTA;
        ctx.stroke();
        ctx.fillStyle = SEAMARK_MAGENTA;
        ctx.font = 'bold ' + Math.round(w * 0.45) + 'px Arial, sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText(letter, w / 2, h / 2 + 0.5);
    }

    // Значок рисуется по имени из seamarkIcon: sm-<вид>[-<форма>][-<цвета>].
    function addSeamarkIcon(name) {
        if (map.hasImage(name)) {
            return;
        }
        var parts = name.split('-');
        var kind = parts[1];
        var tall = kind === 'buoy' || kind === 'beacon';
        var w = tall ? 16 : 20;
        var h = tall ? 22 : 20;
        var ratio = window.devicePixelRatio || 1;
        var canvas = document.createElement('canvas');
        canvas.width = Math.round(w * ratio);
        canvas.height = Math.round(h * ratio);
        var ctx = canvas.getContext('2d');
        ctx.scale(ratio, ratio);
        ctx.lineJoin = 'round';
        ctx.lineCap = 'round';
        if (kind === 'anchor') {
            drawAnchor(ctx, w, h, SEAMARK_MAGENTA);
        } else if (kind === 'harbour') {
            drawAnchor(ctx, w, h, '#1565c0');
        } else if (kind === 'light') {
            drawLight(ctx, w, h, parts[2]);
        } else if (kind === 'buoy') {
            drawBuoy(ctx, w, h, parts[2], parts[3]);
        } else if (kind === 'beacon') {
            drawBeacon(ctx, w, h, parts[2]);
        } else if (kind === 'wreck') {
            drawWreck(ctx, w, h);
        } else if (kind === 'pilot') {
            drawLetterInCircle(ctx, w, h, 'P');
        } else if (kind === 'radio') {
            drawLetterInCircle(ctx, w, h, 'R');
        } else {
            drawDanger(ctx, w, h);
        }
        map.addImage(name, ctx.getImageData(0, 0, canvas.width, canvas.height), { pixelRatio: ratio });
    }

    // ---------- Мост команд 1С ----------

    // Ответ на ready и на status: «<идентификатор набора>|<число пакетов>», пустая строка — набора нет.
    function readyMessage() {
        return cacheSet ? cacheSet.id + '|' + cacheSet.packages : '';
    }

    function applyCommand(command) {
        var op = command && command.op;
        // До startMap список только запоминается (источников ещё нет), как снимок в updateVessels.
        if (op === 'setWatchlist') {
            watchlist = command.watchlist || { version: '', vessels: [] };
            applySnapshot();
            return '';
        }
        if (op === 'updateVessels') {
            snapshot = command.snapshot || null;
            applySnapshot();
            return '';
        }
        if (op === 'addCachePackage') {
            return addCachePackage(command.package);
        }
        if (op === 'startMap') {
            startMap();
            return '';
        }
        if (op === 'status') {
            return readyMessage();
        }
        throw new Error('Неизвестная команда карты: ' + op);
    }

    // data-ack выставляется после раскладки пакета: 1С читает его сразу после click()
    // и следующей командой затирает data-payload.
    function processBridgeCommand(event) {
        if (event) {
            if (event.preventDefault) {
                event.preventDefault();
            }
            if (event.stopPropagation) {
                event.stopPropagation();
            }
        }
        var sequence = commandBridge.getAttribute('data-seq') || '';
        if (!sequence || sequence === processBridgeCommand.lastSequence) {
            return false;
        }
        processBridgeCommand.lastSequence = sequence;
        var payload = commandBridge.getAttribute('data-payload') || '';
        // Иначе строка пакета в полтора мегабайта жила бы в атрибуте DOM до конца сеанса.
        commandBridge.setAttribute('data-payload', '');
        try {
            var result = applyCommand(JSON.parse(payload));
            commandBridge.setAttribute('data-result', result == null ? '' : String(result));
            commandBridge.setAttribute('data-error', '');
        } catch (commandError) {
            commandBridge.setAttribute('data-error', commandError.message || String(commandError));
        }
        commandBridge.setAttribute('data-ack', sequence);
        return false;
    }
    processBridgeCommand.lastSequence = '';

    // ---------- Карта ----------

    function fallbackMessage() {
        return cacheSet
            ? 'Подложка недоступна: показаны только суда'
            : 'Картографические данные не загружены. Администратор: «Загрузка картографических данных»';
    }

    function bindMapEvents() {
        map.on('style.load', function () {
            var fallback = requestedStyle === EMPTY_STYLE;
            requestedStyle = null;
            ensureOwnLayers();
            if (fallback) {
                showMessage(fallbackMessage());
            }
        });

        // Значки обстановки рисуются по требованию: их имена зависят от данных набора.
        map.on('styleimagemissing', function (event) {
            var id = event && event.id;
            var spriteIcon = id ? harborCache.spriteImage(id) : null;
            if (id && id.indexOf('sm-') === 0) {
                addSeamarkIcon(id);
            } else if (spriteIcon) {
                // Значки подложки — из атласа спрайта, который страница декодировала сама.
                if (!map.hasImage(id)) {
                    map.addImage(id, spriteIcon.data, spriteIcon.options);
                }
            } else {
                addVesselIcons();
            }
        });

        map.on('error', function (event) {
            var error = event && event.error;
            var text = error && error.message ? error.message : 'Ошибка карты';
            // Стиль не собрался (например, нет спрайта в наборе): встроенный пустой стиль, чтобы показать хотя бы суда.
            if (requestedStyle && requestedStyle !== EMPTY_STYLE && !event.sourceId) {
                setStyle(EMPTY_STYLE);
                sendEvent('error', 'Подложка недоступна: ' + text);
                return;
            }
            var now = Date.now();
            if (now - lastErrorEventAt >= ERROR_EVENT_INTERVAL_MS) {
                lastErrorEventAt = now;
                sendEvent('error', text);
            }
        });

        map.on('webglcontextlost', function () {
            showMessage('Графический контекст потерян. Перезапустите форму.');
            sendEvent('error', 'Потерян контекст WebGL');
        });

        var clickableLayers = ['vessel-icons', 'vessel-icons-hull-zoom', 'vessel-hulls-fill'];
        clickableLayers.forEach(function (layerId) {
            map.on('click', layerId, function (event) {
                var feature = event.features && event.features[0];
                // Значок и корпус могут перекрываться: один клик открывает popup один раз.
                if (!feature || event.originalEvent.vesselHandled) {
                    return;
                }
                event.originalEvent.vesselHandled = true;
                openPopup(feature.properties.mmsi);
            });
            map.on('mouseenter', layerId, function () {
                map.getCanvas().style.cursor = 'pointer';
            });
            map.on('mouseleave', layerId, function () {
                map.getCanvas().style.cursor = '';
            });
        });

        var mapElement = document.getElementById('map');
        if (typeof ResizeObserver === 'function') {
            new ResizeObserver(function () { map.resize(); }).observe(mapElement);
        } else {
            window.addEventListener('resize', function () { map.resize(); });
        }
    }

    // Карта создаётся из кеша после переноса всех пакетов; без набора — сразу на пустом стиле с судами.
    // Сначала стиль (он же называет спрайт), затем асинхронно атлас спрайта, затем сама карта.
    function startMap() {
        if (map || startMap.pending) {
            return;
        }
        startMap.pending = true;
        var style = cacheSet ? styleFor(ui.basemap) : EMPTY_STYLE;
        harborCache.loadSprite(function () {
            createMap(style);
        });
    }

    function createMap(style) {
        showLoading('');
        seamarkData = buildSeamarkData();
        requestedStyle = style;
        map = new maplibregl.Map({
            container: 'map',
            style: style,
            center: region.center,
            zoom: region.zoom,
            minZoom: region.minZoom,
            maxZoom: region.maxZoom,
            maxBounds: region.maxBounds,
            maxPitch: 0,
            touchPitch: false,
            attributionControl: false
        });
        map.addControl(new maplibregl.NavigationControl({ visualizePitch: false }), 'top-left');
        // Мили — как скорости судов в узлах, километры — для привычной оценки расстояний.
        map.addControl(new maplibregl.ScaleControl({ unit: 'nautical' }), 'bottom-right');
        map.addControl(new maplibregl.ScaleControl({ unit: 'metric' }), 'bottom-right');
        updateAttribution();
        bindMapEvents();
        applySnapshot();
    }

    function sendReady() {
        sendEvent('ready', readyMessage());
    }

    // Набор картографии лежит в кеше страницы: адреса harbor://<ключ> обслуживает обработчик из КешКартыJS.
    harborCache.register(region);
    rebuildVesselData();
    buildLegend();
    updateStaleBanner();
    bindPanel();
    renderWatchPanel();
    // Без набора переключать нечего: обе подложки дали бы пустой стиль.
    document.getElementById('opt-basemap').disabled = !cacheSet;
    if (cacheSet) {
        showLoading(loadingText());
    }

    // ready — до создания карты: форма по нему начинает переносить пакеты, а карта ждёт их.
    // Отправка после load: событие поля во время разбора страницы 1С может не получить.
    commandBridge.addEventListener('click', processBridgeCommand, false);
    commandBridge.setAttribute('data-ready', 'true');
    if (document.readyState === 'complete') {
        setTimeout(sendReady, 0);
    } else {
        window.addEventListener('load', function () { setTimeout(sendReady, 0); });
    }
