    function emptyCollection() {
        return { type: 'FeatureCollection', features: [] };
    }

    function groupOf(shipType) {
        if (shipType == null) {
            return 'other';
        }
        if (shipType >= 70 && shipType <= 79) { return 'cargo'; }
        if (shipType >= 80 && shipType <= 89) { return 'tanker'; }
        if (shipType >= 60 && shipType <= 69) { return 'passenger'; }
        // 31/32 — судно, ведущее буксировку, 52 — буксир: для карты порта это одна группа.
        if (shipType === 31 || shipType === 32 || shipType === 52) { return 'tug'; }
        if (shipType === 50) { return 'pilot'; }
        if (shipType === 30) { return 'fishing'; }
        if (shipType >= 40 && shipType <= 49) { return 'hsc'; }
        if (shipType === 36 || shipType === 37) { return 'pleasure'; }
        return 'other';
    }

    function ageClassOf(ageMin) {
        if (ageMin <= 10) { return 0; }
        if (ageMin <= 60) { return 1; }
        if (ageMin <= 1440) { return 2; }
        return 3;
    }

    function rotationOf(vessel) {
        if (vessel.heading != null) {
            return vessel.heading;
        }
        if (vessel.cog != null && vessel.sog != null && vessel.sog >= 1) {
            return vessel.cog;
        }
        return null;
    }

    function hasDimensions(vessel) {
        return vessel.toBow != null && vessel.toStern != null && vessel.toPort != null
            && vessel.toStarboard != null && vessel.toBow + vessel.toStern > 0;
    }

    // Пятиугольник корпуса в координатах судна (x — к правому борту, y — к носу), повёрнутый по курсу.
    function hullPolygon(vessel, rot) {
        var a = vessel.toBow;
        var b = vessel.toStern;
        var c = vessel.toPort;
        var d = vessel.toStarboard;
        var shoulder = a - 0.15 * (a + b);
        var local = [[-c, -b], [d, -b], [d, shoulder], [(d - c) / 2, a], [-c, shoulder]];
        var theta = rot * Math.PI / 180;
        var cos = Math.cos(theta);
        var sin = Math.sin(theta);
        var metersPerDegLon = 111320 * Math.cos(vessel.lat * Math.PI / 180);
        var ring = local.map(function (point) {
            var east = point[0] * cos + point[1] * sin;
            var north = -point[0] * sin + point[1] * cos;
            return [vessel.lon + east / metersPerDegLon, vessel.lat + north / 111320];
        });
        ring.push(ring[0]);
        return ring;
    }

    function rebuildVesselData() {
        var points = [];
        var hulls = [];
        vesselsByMmsi = {};
        var vessels = snapshot && snapshot.vessels ? snapshot.vessels : [];
        var receivedAt = snapshot ? Date.parse(snapshot.receivedAt) : NaN;

        vessels.forEach(function (vessel) {
            var rot = rotationOf(vessel);
            var ageMin = Math.max(0, (receivedAt - Date.parse(vessel.seen)) / 60000);
            if (!isFinite(ageMin)) {
                ageMin = 0;
            }
            var group = groupOf(vessel.shipType);
            var hasHull = rot != null && hasDimensions(vessel);
            vesselsByMmsi[vessel.mmsi] = vessel;
            points.push({
                type: 'Feature',
                geometry: { type: 'Point', coordinates: [vessel.lon, vessel.lat] },
                properties: {
                    mmsi: vessel.mmsi,
                    group: group,
                    moving: vessel.sog != null && vessel.sog >= 1,
                    rot: rot,
                    label: vessel.name ? vessel.name : String(vessel.mmsi),
                    ageMin: ageMin,
                    ageClass: ageClassOf(ageMin),
                    hasHull: hasHull
                }
            });
            if (hasHull) {
                hulls.push({
                    type: 'Feature',
                    geometry: { type: 'Polygon', coordinates: [hullPolygon(vessel, rot)] },
                    properties: { mmsi: vessel.mmsi, group: group, ageMin: ageMin, ageClass: ageClassOf(ageMin) }
                });
            }
        });
        vesselData = {
            points: { type: 'FeatureCollection', features: points },
            hulls: { type: 'FeatureCollection', features: hulls }
        };
    }

    function drawVesselShape(ctx, size, shape, color) {
        var s = size;
        ctx.beginPath();
        if (shape === 'arrow') {
            // Нос смотрит вверх: при icon-rotate = 0 судно идёт на север.
            ctx.moveTo(s * 0.5, s * 0.06);
            ctx.lineTo(s * 0.82, s * 0.9);
            ctx.lineTo(s * 0.5, s * 0.72);
            ctx.lineTo(s * 0.18, s * 0.9);
        } else {
            ctx.moveTo(s * 0.5, s * 0.2);
            ctx.lineTo(s * 0.78, s * 0.5);
            ctx.lineTo(s * 0.5, s * 0.8);
            ctx.lineTo(s * 0.22, s * 0.5);
        }
        ctx.closePath();
        ctx.fillStyle = color;
        ctx.fill();
        ctx.lineWidth = Math.max(1, s * 0.07);
        ctx.strokeStyle = '#ffffff';
        ctx.stroke();
    }

    function vesselIconCanvas(shape, color, cssSize, ratio) {
        var canvas = document.createElement('canvas');
        var size = Math.round(cssSize * ratio);
        canvas.width = size;
        canvas.height = size;
        drawVesselShape(canvas.getContext('2d'), size, shape, color);
        return canvas;
    }

    // Растровые иконки без sdf: картинка с жёстким краем при sdf рисуется рвано.
    function addVesselIcons() {
        var ratio = window.devicePixelRatio || 1;
        GROUPS.forEach(function (group) {
            ['arrow', 'diamond'].forEach(function (shape) {
                var name = group.id + '-' + shape;
                if (map.hasImage(name)) {
                    return;
                }
                var canvas = vesselIconCanvas(shape, group.color, 24, ratio);
                var image = canvas.getContext('2d').getImageData(0, 0, canvas.width, canvas.height);
                map.addImage(name, image, { pixelRatio: ratio });
            });
        });
    }

    // ---------- Слои ----------

    function ownLayer(id) {
        return OWN_LAYERS.indexOf(id) !== -1;
    }

    function firstBaseSymbolLayerId() {
        var layers = map.getStyle().layers || [];
        for (var i = 0; i < layers.length; i++) {
            if (layers[i].type === 'symbol' && !ownLayer(layers[i].id)) {
                return layers[i].id;
            }
        }
        return undefined;
    }

    function applyRussianLabels() {
        var layers = map.getStyle().layers || [];
        layers.forEach(function (layer) {
            if (layer.type !== 'symbol' || ownLayer(layer.id)) {
                return;
            }
            var field = map.getLayoutProperty(layer.id, 'text-field');
            if (field && /"name|\{name/.test(JSON.stringify(field))) {
                map.setLayoutProperty(layer.id, 'text-field', ['coalesce', ['get', 'name:ru'], ['get', 'name']]);
            }
        });
    }

    function groupColorExpression() {
        var expression = ['match', ['get', 'group']];
        GROUPS.forEach(function (group) {
            expression.push(group.id, group.color);
        });
        expression.push(GROUP_BY_ID.other.color);
        return expression;
    }

    function ageOpacityExpression(maxOpacity) {
        return ['match', ['get', 'ageClass'],
            0, AGE_OPACITY[0] * maxOpacity,
            1, AGE_OPACITY[1] * maxOpacity,
            2, AGE_OPACITY[2] * maxOpacity,
            AGE_OPACITY[3] * maxOpacity];
    }

    function addLayerIfMissing(layer, beforeId) {
        if (!map.getLayer(layer.id)) {
            map.addLayer(layer, beforeId);
        }
    }

    function setVisibility(layerId, visible) {
        if (map.getLayer(layerId)) {
            map.setLayoutProperty(layerId, 'visibility', visible ? 'visible' : 'none');
        }
    }

    function setFilterIfPresent(layerId, filter) {
        if (map.getLayer(layerId)) {
            map.setFilter(layerId, filter);
        }
    }

    // Видимость, фильтр возраста и выделение — из переменных viewer, поэтому переживают смену стиля.
    function applyLayerState() {
        setVisibility('seamark', ui.seamark);
        setVisibility('vessel-hulls-fill', ui.hulls);
        setVisibility('vessel-hulls-line', ui.hulls);
        setVisibility('vessel-hulls-selected', ui.hulls);
        setVisibility('vessel-labels', ui.labels);

        var ageFilter = ui.maxAgeMin == null ? null : ['<=', ['get', 'ageMin'], ui.maxAgeMin];
        function withAge(filter) {
            if (!ageFilter) {
                return filter;
            }
            return filter ? ['all', ageFilter, filter] : ageFilter;
        }
        setFilterIfPresent('vessel-icons', withAge(null));
        // При выключенных корпусах значок нужен всем судам и на крупном масштабе.
        setFilterIfPresent('vessel-icons-hull-zoom', withAge(ui.hulls ? ['!=', ['get', 'hasHull'], true] : null));
        setFilterIfPresent('vessel-labels', withAge(null));
        setFilterIfPresent('vessel-hulls-fill', withAge(null));
        setFilterIfPresent('vessel-hulls-line', withAge(null));

        var selection = ['==', ['get', 'mmsi'], selectedMmsi];
        setFilterIfPresent('vessel-selected', selection);
        setFilterIfPresent('vessel-hulls-selected', selection);
    }

    // ---------- Снимок ----------

    function escapeHtml(text) {
        return String(text).replace(/[&<>"']/g, function (ch) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[ch];
        });
    }

    // Атрибуция снимка приходит из внешней сети: в контрол она попадает только экранированной.
    function updateAttribution() {
        var text = snapshot && snapshot.attribution ? snapshot.attribution : 'Open Waters AIS';
        if (text === attributionText) {
            return;
        }
        attributionText = text;
        if (attributionControl) {
            map.removeControl(attributionControl);
        }
        attributionControl = new maplibregl.AttributionControl({
            compact: false,
            customAttribution: escapeHtml(text)
        });
        map.addControl(attributionControl, 'bottom-right');
    }

    function updateStaleBanner() {
        var text = '';
        if (!snapshot) {
            text = 'Снимка AIS ещё нет';
        } else if (snapshot.freshestSeen) {
            var ageMin = (Date.parse(snapshot.receivedAt) - Date.parse(snapshot.freshestSeen)) / 60000;
            if (ageMin > 60) {
                text = 'Данные AIS устарели: самая свежая позиция ' + formatAge(ageMin) + ' назад';
            }
        }
        staleBanner.textContent = text;
        staleBanner.hidden = !text;
    }

    function applySnapshot() {
        rebuildVesselData();
        var points = map.getSource('vessels');
        if (points) {
            points.setData(vesselData.points);
        }
        var hulls = map.getSource('vessel-hulls');
        if (hulls) {
            hulls.setData(vesselData.hulls);
        }
        updateAttribution();
        updateStaleBanner();
        refreshPopup();
    }

    // ---------- Popup ----------

    function formatAge(ageMin) {
        var minutes = Math.max(0, Math.floor(ageMin));
        if (minutes < 60) {
            return minutes + ' мин';
        }
        var hours = Math.floor(minutes / 60);
        if (hours < 24) {
            return hours + ' ч ' + (minutes % 60) + ' мин';
        }
        return Math.floor(hours / 24) + ' сут ' + (hours % 24) + ' ч';
    }

    function formatNumber(value, digits, unit) {
        return value == null ? '—' : Number(value).toFixed(digits) + (unit || '');
    }

    function popupRow(container, label, value) {
        var row = document.createElement('div');
        row.className = 'vessel-popup-row';
        var labelEl = document.createElement('div');
        labelEl.className = 'vessel-popup-label';
        labelEl.textContent = label;
        var valueEl = document.createElement('div');
        valueEl.className = 'vessel-popup-value';
        valueEl.textContent = value == null || value === '' ? '—' : String(value);
        row.appendChild(labelEl);
        row.appendChild(valueEl);
        container.appendChild(row);
    }

    // Содержимое собирается через textContent: имена и назначения приходят из внешней сети и содержат <, >, &.
    function popupContent(vessel) {
        var root = document.createElement('div');
        root.className = 'vessel-popup';
        var title = document.createElement('div');
        title.className = 'vessel-popup-title';
        title.textContent = vessel.name || ('MMSI ' + vessel.mmsi);
        root.appendChild(title);

        var group = GROUP_BY_ID[groupOf(vessel.shipType)];
        var receivedAt = Date.parse(snapshot.receivedAt);
        var seenAt = Date.parse(vessel.seen);
        popupRow(root, 'IMO / MMSI', (vessel.imo || '—') + ' / ' + vessel.mmsi);
        popupRow(root, 'Позывной', vessel.callsign);
        popupRow(root, 'Флаг', vessel.flag);
        popupRow(root, 'Тип', group.label + (vessel.shipType != null ? ' (код ' + vessel.shipType + ')' : ''));
        popupRow(root, 'Статус', vessel.navStatus != null ? NAV_STATUS[vessel.navStatus] : null);
        popupRow(root, 'Скорость / курс',
            formatNumber(vessel.sog, 1, ' уз') + ' · COG ' + formatNumber(vessel.cog, 0, '°')
            + ' · HDG ' + formatNumber(vessel.heading, 0, '°'));
        popupRow(root, 'Размеры', vessel.length != null || vessel.beam != null
            ? formatNumber(vessel.length, 0, '') + ' × ' + formatNumber(vessel.beam, 0, ' м') : null);
        popupRow(root, 'Осадка', vessel.draught != null ? formatNumber(vessel.draught, 1, ' м') : null);
        popupRow(root, 'Назначение', vessel.destination);
        popupRow(root, 'ETA (UTC)', vessel.eta);
        popupRow(root, 'Позиция от', new Date(seenAt).toLocaleString('ru-RU')
            + ' (' + formatAge((receivedAt - seenAt) / 60000) + ' назад)');
        popupRow(root, 'Источник', vessel.source);
        return root;
    }

    function selectVessel(mmsi) {
        selectedMmsi = mmsi;
        applyLayerState();
    }

    function openPopup(mmsi) {
        var vessel = vesselsByMmsi[mmsi];
        if (!vessel) {
            return;
        }
        if (popup) {
            popup.remove();
        }
        selectVessel(mmsi);
        popup = new maplibregl.Popup({ closeButton: true, closeOnClick: true, maxWidth: '340px', offset: 14 })
            .setLngLat([vessel.lon, vessel.lat])
            .setDOMContent(popupContent(vessel))
            .addTo(map);
        var opened = popup;
        opened.on('close', function () {
            if (popup === opened) {
                popup = null;
                selectVessel(NO_SELECTION);
            }
        });
    }

    // Popup хранит свои координаты и за setData не едет: переносим или закрываем его сами.
    function refreshPopup() {
        if (!popup || selectedMmsi === NO_SELECTION) {
            return;
        }
        var vessel = vesselsByMmsi[selectedMmsi];
        if (vessel) {
            popup.setLngLat([vessel.lon, vessel.lat]).setDOMContent(popupContent(vessel));
        } else {
            popup.remove();
        }
    }

    // ---------- Мост команд 1С ----------

    function applyCommand(command) {
        if (!command || command.op !== 'updateVessels') {
            throw new Error('Неизвестная команда карты: ' + (command && command.op));
        }
        snapshot = command.snapshot || null;
        applySnapshot();
    }

    function processBridgeCommand(event) {
        if (event) {
            if (event.preventDefault) {
                event.preventDefault();
            }
            if (event.stopPropagation) {
                event.stopPropagation();
            }
        }
        var sequence = commandBridge.getAttribute('data-seq') || '';
        if (!sequence || sequence === processBridgeCommand.lastSequence) {
            return false;
        }
        processBridgeCommand.lastSequence = sequence;
        try {
            applyCommand(JSON.parse(commandBridge.getAttribute('data-payload') || ''));
            commandBridge.setAttribute('data-error', '');
        } catch (commandError) {
            commandBridge.setAttribute('data-error', commandError.message || String(commandError));
        }
        commandBridge.setAttribute('data-ack', sequence);
        return false;
    }
    processBridgeCommand.lastSequence = '';

        var fallback = requestedStyle === EMPTY_STYLE;

            showMessage('Подложка недоступна: показаны только суда');

            readySent = true;

        }

    });

    map.on('error', function (event) {
        var error = event && event.error;
        var text = error && error.message ? error.message : 'Ошибка карты';
        // Стиль не загрузился: переходим на встроенный пустой стиль, чтобы показать хотя бы суда.
        if (requestedStyle && requestedStyle !== EMPTY_STYLE && !event.sourceId) {
            setStyle(EMPTY_STYLE);
            sendEvent('error', 'Подложка недоступна: ' + text);
            return;
        }
        var now = Date.now();
        if (now - lastErrorEventAt >= ERROR_EVENT_INTERVAL_MS) {
            lastErrorEventAt = now;
            sendEvent('error', text);
        }
    });

    map.on('webglcontextlost', function () {
        showMessage('Графический контекст потерян. Перезапустите форму.');
        sendEvent('error', 'Потерян контекст WebGL');
    });

    var clickableLayers = ['vessel-icons', 'vessel-icons-hull-zoom', 'vessel-hulls-fill'];
    clickableLayers.forEach(function (layerId) {
        map.on('click', layerId, function (event) {
            var feature = event.features && event.features[0];
            // Значок и корпус могут перекрываться: один клик открывает popup один раз.
            if (!feature || event.originalEvent.vesselHandled) {
                return;
            }
            event.originalEvent.vesselHandled = true;
            openPopup(feature.properties.mmsi);
        });
        map.on('mouseenter', layerId, function () {
            map.getCanvas().style.cursor = 'pointer';
        });
        map.on('mouseleave', layerId, function () {
            map.getCanvas().style.cursor = '';
        });
    });

    }

    commandBridge.addEventListener('click', processBridgeCommand, false);
